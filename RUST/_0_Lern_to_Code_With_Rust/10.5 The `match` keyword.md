    Cold,

            LaundryCycle::Cold => {
            println!("Running the laudry with cold temp");
        }
        LaundryCycle::Hot{ temp} =>{
            println!("Running the laundry with a temp of {temp}");
        }
        LaundryCycle::Delicate(fabric_type) =>{
            println!("Running the laundry with a delicate cycle for {fabric_type}");
        }
        }
    }
}

fn main() {
    LaundryCycle::Cold.wash_laundry(); // Running the laudry with cold temp

    let hot = LaundryCycle::Hot{temp: 100}; 
    hot.wash_laundry();// Running the laundry with a temp of 100
    
    LaundryCycle::Delicate(String::from("Silk")).wash_laundry(); // Running the laundry with a delicate cycle for Silk

    
}

# March with exact value
```rust

#[derive(Debug)]
enum Milk{
    LowFat(i32),
    Whole,
}

impl Milk{
    fn drink(self){
        match self{
            Milk::LowFat(2) => {
                println!("2% is good")
            }
            Milk::LowFat(percent) =>{
                println!("your {percent:?} %")
            }
            Milk::Whole =>{
                println!("Whole milk") ;
            }
            
        }
    }
}


fn main(){
    Milk::LowFat(1).drink(); // your 1 %
    Milk::LowFat(2).drink(); // 2% is good
    Milk::LowFat(12).drink(); // your 12 %
    Milk::Whole.drink(); // Whole milk
    

}```

# The `if let ` Construct 
#rust/if_let

>[!info] `if let`
> The `if let` construct combines an if statement with a variable declaration.
>


```rust

#[derive(Debug)]
enum Milk{
    LowFat(i32),
    Whole,
    NonDairy {kind: String},
}


fn main(){
    let my_beverage: Milk = Milk::NonDairy{kind: String::from("Oat")};
    
    if let Milk::NonDairy{kind} = my_beverage {
        println!("You have nonDairy {kind:?} milk") ;//You have nonDairy "Oat" milk
    }
    

}

```
